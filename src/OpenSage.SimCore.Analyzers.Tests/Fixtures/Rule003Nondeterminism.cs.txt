using System.Diagnostics;

namespace Fixture

    {

        public Guid Id() => Guid.NewGuid();                     // EXPECT: SIMCORE003

        public int Uptime() => Environment.TickCount;           // EXPECT: SIMCORE003

        public int Cores() => Environment.ProcessorCount;       // EXPECT: SIMCORE003
    }
}
